# title: בקרה — עיתויים, סוגים ובנצ'מרקינג
# שיעור 8. כל עובדה כאן מופיעה בבנק השאלות bank8.js או במילון (src/glossary/L8.md).
# ש = המראיינת (Hila), א = המסביר (Avri). מספרים במילים, בלי מקפים, גרש אסקי.

ש: פֶּרֶק חֲמִישִׁי: בַּקָרָה. שִׁעוּר שְׁמוֹנָה. נַתְחִיל בַּהַגְדָרָה.

א: תַהֲלִיךְ הַבַּקָרָה הוּא תַהֲלִיךְ דִינָמִי וְשִׁיטָתִי, בִּשְׁלוֹשָׁה שְׁלַבִּים. קוֹבְעִים סְטַנְדַרְטִים בְּרוּרִים לְבִצוּעִים. מַשְׁוִים אֶת הַבִּצוּעַ בְּפֹעַל לַבִּצוּעַ הָרָצוּי. וְנוֹקְטִים פְּעֻלָה מְתַקֶנֶת. וְהַמַטָרָה, לְדִבְרֵי הַמַרְצָה, הִיא לִלְמֹד וּלְהִשְׁתַפֵּר מִתְקוּפָה לִתְקוּפָה.

ש: וּלְפִי מָה קוֹבְעִים סְטַנְדַרְט?

א: לְפִי שְׁלוֹשָׁה דְבָרִים. מָה הַלָקוֹחוֹת רוֹצִים. מָה הַמִתְחָרִים נוֹתְנִים. וּמָה הִשַׂגְנוּ בַּתְקוּפָה הַקוֹדֶמֶת.

ש: וְלָמָה הַמִתְחָרִים כֹּל כָּךְ חֲשׁוּבִים?

א: הַדֻגְמָה שֶׁל הַמַרְצָה: אִם הַמִתְחָרִים מְבִיאִים פִּיצָה תוֹךְ רֶבַע שָׁעָה, וַאֲנִי תוֹךְ חֲצִי שָׁעָה, אֵין לִי זְכוּת קִיוּם.

ש: וְיֵשׁ בַּקָרָה גְלוּיָה וּבַקָרָה סְמוּיָה.

א: וְהַמַרְצָה אָמְרָה: הַבַּקָרָה הַסְמוּיָה הִיא הַבַּקָרָה הָאֲמִתִית. לְמָשָׁל, זוּג שֶׁמִתְאָרֵחַ בְּמָלוֹן כְּאוֹרְחִים רְגִילִים וּמְדַוֵחַ, אוֹ צֶוֶת שֶׁמְנַתֵחַ שִׂיחוֹת בַּבַּנְק. בְּבַקָרָה גְלוּיָה, כְּשֶׁיוֹדְעִים יוֹם מֵרֹאשׁ עַל בִּקֹרֶת הַפֶּתַע, רוֹאִים מַשֶׁהוּ שֶׁאֵין לוֹ קֶשֶׁר לְיוֹם רָגִיל.

ש: הַדֻגְמָה מֵהַצָבָא הָאָמֵרִיקָאִי: הַתַחְקִיר.

א: הַמִלָה תַחְקִיר הִיא מִלָה לַהֲבָנָה וְלִלְמִידָה. הַתַחְקִיר מֵבִיא שְׁקִיפוּת, תַהֲלִיכֵי לְמִידָה, אִרְגוּן לוֹמֵד, וּבִטָחוֹן פְּסִיכוֹלוֹגִי.

ש: וְעַכְשָׁו מֻשָׂג שֶׁהַמַרְצָה אָמְרָה שֶׁתִצְטָרְכוּ גַם לְמִבְחָן.

א: בֶּנְצְ'מַרְקִינְג. מַשְׁוִים בֵּין יְחִידוֹת מַקְבִּילוֹת לְפִי מַדָדִים מְשֻׁתָפִים, בִּשְׁקִיפוּת, וְזֶה יוֹצֵר תַחֲרוּת בּוֹנָה בֵּינֵיהֶן. הַדֻגְמָה: מְנַהֶלֶת בֵּית חוֹלִים שֶׁמַשְׁוָה אֶת הַשֵׁרוּת בֵּין הַמַחְלָקוֹת, בְּאָדֹם, צָהֹב וְיָרֹק.

ש: וּמִי בּוֹנֶה אֶת לוּחַ הַמַדָדִים?

א: הַמְנַהֵל, יַחַד עִם הַמְנַהֲלִים שֶׁלוֹ, כְּמוֹ בְּנִהוּל עַל פִּי יְעָדִים. וּבוֹחֲרִים בּוֹ רַק אֶת הַמַדָדִים הָרֵלֵוַנְטִיִים וְהַקְרִיטִיִים.

ש: הַפְּעֻלָה הַמְתַקֶנֶת.

א: שְׁלוֹשָׁה צְעָדִים לְפִי הַשֶׁקֶף. מְזַהִים אֶת הַסְטִיוֹת בַּבִּצוּעִים. מְנַתְחִים אוֹתָן. וּמְפַתְחִים וּמְיַשְׂמִים תֹכְנִיוֹת לְתִקוּן.

ש: וּמָה חָשׁוּב בַּנִתוּחַ?

א: לִבְדֹק לָעֹמֶק אֶת הַסִבָּה. בְּרֶשֶׁת פִּיצָה, לְמָשָׁל: הָעוֹבְדִים, הַמוֹקֵד הַטֵלֵפוֹנִי, הַתַנוּרִים אוֹ הַשְׁלִיחִים. וּלְתַקֵן עוֹד בְּאוֹתוֹ רִבְעוֹן, לֹא לְחַכּוֹת לָרִבְעוֹן הַבָּא.

ש: וְעוֹד כְּלָל עַל מַדָדִים.

א: לְוַדֵא שֶׁהַמַדָדִים לֹא מִתְנַגְשִׁים זֶה בָּזֶה, וּלְשַׁלֵב מַדָד כַּמוּתִי עִם מַדָד אֵיכוּתָנִי. הַמַדָד הַכַּמוּתִי יַרְאֶה כַּמָה נִמְכַּר, וְהָאֵיכוּתִי יַרְאֶה אִם הַנָצִיג הָיָה נֶחְמָד.

ש: עַכְשָׁו שְׁלֹשֶׁת הָעִתוּיִים שֶׁל הַבַּקָרָה. הָרִאשׁוֹן.

א: בַּקָרָה לְאַחַר מַעֲשֶׂה, פִידְבֶּק. אוֹסְפִים מֵידָע עַל לִקוּיִים אַחֲרֵי שֶׁקָרוּ. הִיא נוֹתֶנֶת תְמוּנָה מְלֵאָה, אֲבָל אֶת מָה שֶׁקָרָה כְּבָר אִי אֶפְשָׁר לְהָשִׁיב. הַגִיחָה נִגְמְרָה.

ש: הַשֵׁנִי.

א: בַּקָרָה בְּמַקְבִּיל, קוֹנְקָרֶנְט. אוֹסְפִים מֵידָע עַל לִקוּיִים בִּזְמַן שֶׁהֵם קוֹרִים. הִיא חֲשׁוּבָה כִּי מְטַפְּלִים לִפְנֵי שֶׁהַלָקוֹחַ עוֹזֵב, וְלִפְנֵי שֶׁהַתַקָלָה הוֹפֶכֶת לְמַשְׁבֵּר.

ש: וְיֵשׁ לָה מְחִיר?

א: כֵּן. בְּחֶבְרוֹת שֵׁרוּת הִיא יְקָרָה מְאוֹד. בַּמָלוֹן יֵשׁ מֵאוֹת אִינְטֶרְאַקְצִיוֹת בְּכֹל דַקָה. וּבְיִצוּר קַל יוֹתֵר, כִּי הַכֹּל בְּמִסְפָּרִים. וְיֵשׁ סַכָּנָה: כְּשֶׁהַבַּקָרָה הוֹפֶכֶת לְמַטָרָה בִּמְקוֹם לִכְלִי. בַּמוֹקְדִים, הַנְצִיגִים מְמַהֲרִים לִסְגֹר שִׂיחוֹת, כִּי מוֹדְדִים אוֹתָם עַל זֶה.

ש: וְהַשְׁלִישִׁי.

א: בַּקָרָה מַקְדִימָה, פִיד פוֹרְוַורְד. מְנַטְרִים אֶת הַתְשׂוּמוֹת וְלֹא אֶת הַתְפוּקוֹת, כְּדֵי לִמְנֹעַ לִקוּיִים לִפְנֵי שֶׁהֵם קוֹרִים. הַדֻגְמָה: אֶפֶּל, שֶׁבּוֹדֶקֶת אֶת הַסַפָּקִים שֶׁלָה, כְּדֵי שֶׁרְכִיב פָּגוּם לֹא יַגִיעַ לַלָקוֹחַ.

ש: וּמָה הַמְחִיר שֶׁלָה?

א: הִיא יְקָרָה, וְהִיא מְעוֹרֶרֶת אַנְטָגוֹנִיזְם אֵצֶל הַסַפָּקִים. אֲבָל הִיא מוֹנַעַת אֶת רֹב הַתַקָלוֹת. וּבַמִלִים שֶׁל הַמַרְצָה: חֶבְרוֹת שֶׁחוֹסְכוֹת בְּבַקָרָה מַקְדִימָה, מְשַׁלְמוֹת עָלֶיהָ אַחַר כָּךְ. וְצָרִיךְ לִשְׁקֹל עֲלוּת מוּל תוֹעֶלֶת.

ש: וְאָמָזוֹן מַדְגִימָה אֶת שְׁלָשְׁתָם.

א: לְאַחַר מַעֲשֶׂה: בּוֹדְקִים אִם הַסְחוֹרָה הִגִיעָה בַּזְמַן. בְּמַקְבִּיל: תוֹךְ כְּדֵי הָעֲבוֹדָה שֶׁל הָעוֹבְדִים. וּמַקְדִימָה: עוֹד לִפְנֵי שֶׁהַסְחוֹרָה נֶאֱרֶזֶת.

ש: דֻגְמָאוֹת לְבַקָרָה: מְלַאי וְתַקְצִיב.

א: בְּבַקָרַת מְלַאי יֵשׁ שְׁלוֹשָׁה סוּגִים. מוּצָרִים גְמוּרִים, שֶׁהֵם הַיְקָרִים בְּיוֹתֵר, כִּי כְּבָר הֻשְׁקְעוּ בָּהֶם עֲבוֹדָה וַעֲלוּיוֹת, וְיֵשׁ לְשַׁלֵם עַל הָאִחְסוּן שֶׁלָהֶם. מְלַאי בַּתַהֲלִיךְ, כְּמוֹ מְנוֹעִים שֶׁמְחַכִּים לְהַרְכָּבָה. וְחָמְרֵי גֶלֶם, שֶׁהֵם זוֹלִים יוֹתֵר.

א: וּבַקָרָה תַקְצִיבִית הִיא מֵהַשִׁיטוֹת הַנְפוֹצוֹת בְּיוֹתֵר: קוֹבְעִים יְעָדִים לְהוֹצָאוֹת, עוֹקְבִים וּמַשְׁוִים לַתַקְצִיב, וּמְשַׁנִים לְפִי הַצֹרֶךְ.

ש: מָה מְאַפְיֵן מַעֲרֶכֶת בַּקָרָה אֵפֶקְטִיבִית?

א: שִׁשָׁה דְבָרִים בַּשֶׁקֶף. הַמֵידָע מוּבָן, וּמֻצָג בְּצוּרָה מַתְאִימָה. הוּא מַגִיעַ לַמְנַהֲלִים שֶׁאַחְרָאִים עַל הַתְחוּם. הוּא מְדַוֵחַ עַל סְטִיוֹת מַהֵר כְּכֹל הָאֶפְשָׁר, וְרָצוּי עוֹד לִפְנֵי שֶׁהֵן קוֹרוֹת.

א: הַמַעֲרֶכֶת מִתְמַקֶדֶת בַּפְּעִילוּיוֹת הַקְרִיטִיוֹת לְהַצְלָחָה. הַמֵידָע לֹא מֻשְׁפָּע מִגוֹרְמִים שֶׁאֵינָם קְשׁוּרִים לְמַטְרַת הַבַּקָרָה. וְהַמַעֲרֶכֶת עִקְבִית עִם הַמַטָרָה שֶׁל הַפְּעִילוּת.

ש: וּמָה לֹא נִמְצָא בָּרְשִׁימָה?

א: כַּמָה שֶׁיוֹתֵר בַּקָרוֹת עַל כֹּל פְּעִילוּת. לְהֶפֶךְ: בַּקָרוֹת רַבּוֹת עַל פְּעִילוּיוֹת שׁוּלִיוֹת הֵן לֹא כַּלְכָּלִיוֹת, וְגוֹזְלוֹת זְמַן.

ש: וְשָׁעוֹן הַנוֹכְחוּת?

א: הַנוֹכְחוּת לֹא מְשַׁקֶפֶת אֶת הַתְפוּקָה. הִיא רֵלֵוַנְטִית בְּיִצוּר וּבְתַחֲנוֹת דֶלֶק, וּפָחוֹת בְּהַיְטֵק, שֶׁבּוֹ רוֹצִים לִרְאוֹת תְפוּקָה.

ש: חֲמֵשֶׁת סוּגֵי הַבַּקָרָה. הָרִאשׁוֹן: בַּקָרָה בְּיוּרוֹקְרָטִית.

א: בַּקָרָה מִלְמַעְלָה לְמַטָה, בְּעֶזְרַת תַגְמוּלִים וָעֳנָשִׁים, מְדִינִיוּת וּכְלָלִים. לְעִתִים קְרוֹבוֹת הִיא לֹא יְעִילָה, וּמְאוֹד עֲמִידָה לְשִׁנוּי.

ש: הַשֵׁנִי: בַּקָרָה אוֹבְּיֶקְטִיבִית.

א: מִשְׁתַמְשִׁים בְּמַדָדִים נִצְפִּים שֶׁל הַהִתְנַהֲגוּת שֶׁל הָעוֹבְדִים, אוֹ שֶׁל הַתְפוּקוֹת שֶׁלָהֶם. יֵשׁ בַּקָרַת הִתְנַהֲגוּת, וְיֵשׁ בַּקָרַת תְפוּקָה בְּעֶזְרַת תַגְמוּלִים וְתַמְרִיצִים. וְהַמַרְצָה הִדְגִישָׁה לְשַׁלֵב בֵּין הַשְׁתַיִם. עוֹבֵד חָדָשׁ וּמִקְצוֹעִי עָשׂוּי לְהַרְאוֹת תְפוּקָה נְמוּכָה בַּהַתְחָלָה, וְעוֹבֵד שֶׁמֻכָּר הַרְבֵּה עָשׂוּי לִהְיוֹת לֹא נָעִים לְלָקוֹחוֹת.

ש: הַשְׁלִישִׁי: בַּקָרָה נוֹרְמָטִיבִית.

א: הִיא נוֹצֶרֶת מִמִיוּן קַפְּדָנִי שֶׁל עוֹבְדִים, וּמֵהִתְבּוֹנְנוּת וְהַקְשָׁבָה לָעוֹבְדִים הַוָתִיקִים. הַמַרְצָה קוֹרֵאת לָה בַּקָרַת דִי אֶן אֵיי: הַנוֹרְמוֹת מְעַצְבוֹת אֶת הָאִרְגוּן, וְלָכֵן צָרִיךְ פָּחוֹת נְהָלִים. הַדֻגְמָאוֹת: בְּגוּגֶל הַצֶוֶת לֹא הוֹלֵךְ הַבַּיְתָה עַד שֶׁהַפְּרוֹיֵקְט הִסְתַיֵם, וּבְנוֹרְדְסְטְרוֹם מְבִיאִים לַלָקוֹחַ אֶת הַפְּרִיט, גַם מֵחֲנוּת מִתְחָרֶה.

ש: הָרְבִיעִי: בַּקָרָה קוֹנְצֶרְטִיבִית.

א: וִסוּת הַהִתְנַהֲגוּת וְהַהַחְלָטוֹת שֶׁל הָעוֹבְדִים, בְּעֶזְרַת הָעֲרָכִים וְהָאֱמוּנוֹת שֶׁל קְבוּצַת הָעֲבוֹדָה. בִּקְבוּצוֹת אוֹטוֹנוֹמִיוֹת, הַחֲבֵרִים עַצְמָם מְבַקְרִים אֶת הַתַהֲלִיכִים, הַתְפוּקוֹת וְהַהִתְנַהֲגוּת. הַמָשָׁל שֶׁל הַמַרְצָה: הַמְנַצֵחַ בּוֹדֵק אֶת הַתִזְמֹרֶת כֻּלָה, וְלֹא כֹּל נַגָן בְּנִפְרָד.

ש: וְהַחִסָרוֹן שֶׁלָה?

א: הִיא עֲלוּלָה לְעוֹדֵד טְרֶמְפִּיסְטִים, וּלְהוֹרִיד אֶת הַמוֹטִיבַצְיָה שֶׁל הַטוֹבִים בְּיוֹתֵר.

ש: וְהַחֲמִישִׁי: בַּקָרָה עַצְמִית.

א: הָעוֹבְדִים מְבַקְרִים אֶת הַהִתְנַהֲגוּת שֶׁל עַצְמָם, בְּתוֹךְ גְבוּלוֹת מֻגְדָרִים. הֵם מַצִיבִים לְעַצְמָם יְעָדִים וְעוֹקְבִים אַחֲרֵי הַהִתְקַדְמוּת, וְהַמְנַהֲלִים מְלַמְדִים אוֹתָם אֶת הַכִּשׁוּרִים. בַּמַהוּת, זֶה נִהוּל עַצְמִי.

ש: וְהַמַרְצָה אָמְרָה עָלֶיהָ מַשֶׁהוּ מְיֻחָד.

א: שֶׁהִיא הֲכִי חֲשׁוּבָה. הִיא מַתְאִימָה לְאִרְגוּנִים שְׁטוּחִים וּלְצִמְצוּם, וְהַמְנַהֲלִים מִתְפַּנִים לַחֲשֹׁב עַל אַסְטְרָטֶגְיָה, כִּי הֵם סוֹמְכִים עַל הָעוֹבֵד.

ש: שְׁאֵלַת דִיוּן: הַאִם תַרְבּוּת חֲזָקָה יְכוֹלָה לְהַחֲלִיף אֶת הַבַּקָרָה הַפוֹרְמָלִית?

א: בְּמִדָה רַבָּה, כֵּן. תַרְבּוּת חֲזָקָה מְגַדֶלֶת עוֹבְדִים וּמְנַהֲלִים שֶׁמְבַקְרִים אֶת עַצְמָם.

ש: מַעֲרֶכֶת הַבַּקָרָה שֶׁל הַייֶר.

א: הַעֲרָכָה יוֹמִית שֶׁל עוֹבְדֵי הַמִפְעָל. מוֹדֵל יִחוּדִי לְגִיוּס וּלְקִדוּם. תַמְרִיצִים לִמְנַהֲלִים עִם בִּצוּעִים גְבוֹהִים. דוּחַ רֶוַח וְהֶפְסֵד חָדְשִׁי וְאִישִׁי לְכֹל עוֹבֵד. וְעֶקְרוֹן שְׁמוֹנִים עֶשְׂרִים.

ש: מָה פֵּרוּשׁ עִקָרוֹן שְׁמוֹנִים עֶשְׂרִים?

א: עֶשְׂרִים אָחוּז מֵהָעוֹבְדִים, שֶׁהֵם הַמְנַהֲלִים, אַחְרָאִים לִשְׁמוֹנִים אָחוּז מִתוֹצְאוֹת הַבִּצוּעִים שֶׁל הַחֶבְרָה.

ש: וְנֶטְפְלִיקְס?

א: יֵשׁ לָה מַעֲרֶכֶת בַּקָרָה פִינַנְסִית, נִהוּל תֹכֶן, תַשְׁתִית טֶכְנוֹלוֹגִית, תְמִיכַת לָקוֹחוֹת, וּבַקָרָה הִתְנַהֲגוּתִית.

ש: וּלְסִיוּם, בַּקָרָה הִירַרְכִית מוּל בַּקָרָה מְבֻזֶרֶת.

א: בַּקָרָה הִירַרְכִית מִשְׁתַמֶשֶׁת בִּכְלָלִים וּבִנְהָלִים מְפֹרָטִים, בְּסַמְכוּת מִלְמַעְלָה וּבִמְפַקְחֵי אֵיכוּת. הִיא מַדְגִישָׁה תַגְמוּלִים חִיצוֹנִיִים, וְהַתַרְבּוּת בָּה נֻקְשָׁה.

א: בַּקָרָה מְבֻזֶרֶת מִשְׁתַמֶשֶׁת בִּמְעַט כְּלָלִים, וְנִשְׁעֶנֶת עַל עֲרָכִים וְעַל בַּקָרָה קְבוּצָתִית וְעַצְמִית. הַמִבְנֶה שָׁטוּחַ, וְכֻלָם מְפַקְחִים עַל הָאֵיכוּת. יֵשׁ בָּה תַגְמוּלִים חִיצוֹנִיִים וּפְנִימִיִים, הָעוֹבְדִים לוֹקְחִים יֹזְמָה, וְהַתַרְבּוּת מִסְתַגֶלֶת.

ש: שְׁאֵלוֹת לַדֶרֶךְ. רִאשׁוֹנָה: מָהֵם שְׁלֹשֶׁת הַשְׁלַבִּים שֶׁל תַהֲלִיךְ הַבַּקָרָה?

~4

א: קְבִיעַת סְטַנְדַרְטִים, הַשְׁוָאַת הַבִּצוּעַ בְּפֹעַל לְרָצוּי, וּפְעֻלָה מְתַקֶנֶת.

ש: שְׁנִיָה: לְפִי מָה קוֹבְעִים סְטַנְדַרְט?

~4

א: לְפִי מָה שֶׁהַלָקוֹחוֹת רוֹצִים, מָה שֶׁהַמִתְחָרִים נוֹתְנִים, וּמָה שֶׁהִשַׂגְנוּ בַּתְקוּפָה הַקוֹדֶמֶת.

ש: שְׁלִישִׁית: אֵיזוֹ בַּקָרָה מַנְטְרַת אֶת הַתְשׂוּמוֹת, כְּדֵי לִמְנֹעַ לִקוּיִים לִפְנֵי שֶׁהֵם קוֹרִים?

~4

א: בַּקָרָה מַקְדִימָה, פִיד פוֹרְוַורְד.

ש: רְבִיעִית: מָה זֶה בֶּנְצְ'מַרְקִינְג?

~4

א: הַשְׁוָאָה בֵּין יְחִידוֹת מַקְבִּילוֹת לְפִי מַדָדִים מְשֻׁתָפִים, שֶׁיוֹצֶרֶת תַחֲרוּת בּוֹנָה בֵּינֵיהֶן.

ש: חֲמִישִׁית: וִסוּת הַהִתְנַהֲגוּת בְּעֶזְרַת הָעֲרָכִים וְהָאֱמוּנוֹת שֶׁל קְבוּצַת הָעֲבוֹדָה. אֵיזֶה סוּג בַּקָרָה זֶה?

~4

א: בַּקָרָה קוֹנְצֶרְטִיבִית.

ש: וְאַחֲרוֹנָה: מָה אוֹמֵר עִקָרוֹן שְׁמוֹנִים עֶשְׂרִים שֶׁל הַייֶר?

~4

א: שֶׁעֶשְׂרִים אָחוּז מֵהָעוֹבְדִים, הַמְנַהֲלִים, אַחְרָאִים לִשְׁמוֹנִים אָחוּז מֵהַתוֹצָאוֹת.

ש: זֶהוּ לַפֶּרֶק הַזֶה. בַּפֶּרֶק הַבָּא: צְוָתִים. נוֹרְמוֹת וּלְכִידוּת, קוֹנְפְלִיקְטִים, וּשְׁלַבֵּי הַהִתְפַּתְחוּת שֶׁל צֶוֶת.

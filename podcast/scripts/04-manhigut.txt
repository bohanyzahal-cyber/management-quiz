# title: מנהיגות — מהשריג ועד אינטליגנציה רגשית
# שיעור 7. כל עובדה כאן מופיעה בבנק השאלות bank7.js או במילון (src/glossary/L7.md).
# ש = המראיינת (Hila), א = המסביר (Avri). מספרים במילים, בלי מקפים, גרש אסקי.

ש: פֶּרֶק רְבִיעִי: מַנְהִיגוּת. שִׁעוּר שֶׁבַע. וְנַתְחִיל בַּהֶעָרָה שֶׁל הַמַרְצָה עַל הַשִׁעוּר הַזֶה.

א: בַּשִׁעוּר הֻצְגוּ חֲמִשָׁה מוֹדֵלִים שֶׁל מַנְהִיגוּת. שְׁלוֹשָׁה מֵהֶם נִלְמָדִים גַם בְּקוּרְס מַנְהִיגוּת אַחֵר: הַתְכוּנוֹת, הַשָׂרִיג וְנָתִיב מַטָרָה. אֶת שְׁנֵי הָאַחֲרוֹנִים, אֶל אֶם אֶקְס וְאִינְטֵלִיגֶנְצִיָה רִגְשִׁית, הִיא הִגְדִירָה כְּמֶרְכָּזִיִים, אֵלֶה שֶׁהִיא רוֹצָה לְהַעֲבִיר לָכֶם.

ש: לִפְנֵי הַמוֹדֵלִים: מָה הַהֶבְדֵל בֵּין מְנַהֵל לְמַנְהִיג?

א: לְפִי הַטַבְלָה בַּשֶׁקֶף, הַמַנְהִיגִים עוֹשִׂים אֶת הַדְבָרִים הַנְכוֹנִים, וְהַמְנַהֲלִים עוֹשִׂים אֶת הַדְבָרִים נָכוֹן. הַמַנְהִיגִים מוֹבִילִים שִׁנוּי, וְהַמְנַהֲלִים שׁוֹמְרִים עַל הַקַיָם. הַמַנְהִיגִים מִסְתַכְּלִים לִטְוָח אָרֹךְ, וְהַמְנַהֲלִים לִטְוָח קָצָר.

א: וְעוֹד שְׁנֵי זוּגוֹת. הַמַנְהִיגִים עוֹסְקִים בְּמַטָרוֹת, וְהַמְנַהֲלִים בְּאֶמְצָעִים. הַמַנְהִיגִים הֵם אַדְרִיכָלִים, וְהַמְנַהֲלִים הֵם בַּנָאִים. וְהַמַנְהִיגִים מְנִיעִים וְנוֹתְנִים הַשְׁרָאָה, בִּזְמַן שֶׁהַמְנַהֲלִים פּוֹתְרִים בְּעָיוֹת.

ש: לָמָה בַּנָאִים?

א: כִּי הֵם זְקוּקִים לְאֶמְצָעִים: סַמְכוּת, תַקְצִיב וּמַעֲרָכוֹת. לְעֻמָתָם, הָאַדְרִיכָלִים מְעַצְבִים, מְשַׁנִים וּמַגְדִירִים לְאָן רוֹצִים לְהַגִיעַ.

ש: וְהַדֻגְמָה מֵהַכִּתָה?

א: דַפְּנִי לִיף, שֶׁהוֹבִילָה אֶת מְחָאַת הָאֹהָלִים. לֹא הָיָה לָה כֶּסֶף, לֹא הָיְתָה לָה סַמְכוּת, וְאַף אֶחָד לֹא הִגְדִיר לָה תַפְקִיד. בַּמִלִים שֶׁל הַמַרְצָה: הַמַנְהִיגִים לֹא צְרִיכִים תַקְצִיב, סַמְכוּת אוֹ הַרְשָׁאָה. יֵשׁ לָהֶם חָזוֹן וְאָגֶ'נְדָה.

ש: אָז כֹּל מְנַהֵל הוּא מַנְהִיג?

א: הַתְשׁוּבָה שֶׁל הַמַרְצָה: לֹא.

ש: הַמוֹדֵל הָרִאשׁוֹן: גִישַׁת הַתְכוּנוֹת.

א: שֶׁבַע תְכוּנוֹת. דַחַף. רָצוֹן לְהוֹבִיל. יֹשֶׁר וְיֹשְׁרָה. בִּטָחוֹן עַצְמִי. יַצִיבוּת רִגְשִׁית. יְכֹלֶת קוֹגְנִיטִיבִית. וְיֶדַע בָּעֵסֶק.

ש: וְאֶפְשָׁר לְפַתֵחַ אוֹתָן?

א: הַהַנָחָה שֶׁל הַגִישָׁה הִיא שֶׁהֵן בְּעִקָר מוּלָדוֹת. אֶפְשָׁר לְשַׁפֵּר בִּנְיוּאַנְסִים, אֲבָל לֹא מֵאֶפֶס לְמֵאָה.

ש: וְאֵיזוֹ תְכוּנָה הֲכִי קַל לְשַׁפֵּר הַיוֹם?

א: יֶדַע בָּעֵסֶק. בְּמִלִים שֶׁל הַמַרְצָה: הַיֶדַע שֶׁלִי זֶה הַיֶדַע שֶׁלָךְ, וְהַכִּשָׁרוֹן הוּא לְהִשְׁתַמֵשׁ בַּכְּלִי הַנָכוֹן. וְשִׂימוּ לֵב לַמַלְכֹּדֶת: צַיְתָנוּת לַנְהָלִים אֵינָה אַחַת הַתְכוּנוֹת.

ש: הַמוֹדֵל הַשֵׁנִי: הַשָׂרִיג שֶׁל בְּלֵייק וּמוּטוֹן.

א: שְׁנֵי צִירִים. צִיר אֶחָד הוּא הַדְאָגָה לַמְשִׂימָה, וְהַשֵׁנִי הוּא הַדְאָגָה לָאֲנָשִׁים. עַל הַשָׂרִיג יֵשׁ חֲמִשָׁה סִגְנוֹנוֹת, וְהָרַעֲיוֹן הוּא שֶׁלֹא טוֹב לִהְיוֹת בְּאַף אַחַת מֵהַפִּנוֹת.

ש: תָאֵר אֶת הַפִּנוֹת.

א: מְנַהֵל קַאנְטְרִי קְלַאבּ גָבוֹהַ בִּדְאָגָה לַאֲנָשִׁים וְנָמוּךְ בַּמְשִׂימָה. כֵּיף, חֶבְרֶ'ה, גִבּוּשׁ, אֲבָל מְפַסְפְסִים אֶת הַמְשִׂימָה.

א: מְנַהֵל מְשִׂימָתִי מְאוֹד, שֶׁנָמוּךְ בִּדְאָגָה לַאֲנָשִׁים, עוֹמֵד בְּלוּחוֹת הַזְמַנִים, אֲבָל שׁוֹכֵחַ שֶׁהוּא מְנַהֵל אֲנָשִׁים וְלֹא מְכוֹנוֹת. בַּסוֹף יֵשׁ סְטְרֵס וּשְׁחִיקָה.

א: וְהַנְמוּכִים בִּשְׁנֵי הַצִירִים? בַּסְלֵנְג הַמִקְצוֹעִי קוֹרְאִים לָהֶם עֲצִיצִים: מְנַהֲלִים שֶׁבַּמִקְרֶה הִגִיעוּ.

ש: וְאֵיפֹה רוֹצִים לִהְיוֹת?

א: מֵחָמֵשׁ חָמֵשׁ וְעַד תֵשַׁע תֵשַׁע. לְאַזֵן בֵּין הַמְשִׂימָה לַאֲנָשִׁים. וְכֹל מְנַהֵל צָרִיךְ לָדַעַת אֵיפֹה הוּא חָזָק וְאֵיפֹה הוּא חַלָשׁ, וּלְחַזֵק אֶת הַצַד הַחַלָשׁ בְּהַכְשָׁרוֹת.

ש: וְהַטִיפּ שֶׁל הַמַרְצָה, כְּיוֹעֶצֶת אִרְגוּנִית?

א: קַח סְגָן שֶׁהוּא מֵאָה שְׁמוֹנִים מַעֲלוֹת מִמְךָ. אִם אַתָה מְשִׂימָתִי, הַסְגָן יְאַזֵן אוֹתְךָ בַּצַד הָאֱנוֹשִׁי.

ש: וּמָה הַקֶשֶׁר לַגִישׁוֹת הַהִתְנַהֲגוּתִיוֹת?

א: שְׁנֵי הַצִירִים שֶׁל הַשָׂרִיג מַקְבִּילִים לָהֶן. הִתְחַשְׁבוּת הִיא הִתְנַהֲגוּת שֶׁמְכַוֶנֶת לַאֲנָשִׁים: קֶשֶׁב לִכְפִיפִים, אֵמוּן הֲדָדִי וַעֲבוֹדַת צֶוֶת.

א: מִבְנֶה יוֹזֵם הִיא הִתְנַהֲגוּת שֶׁמְכַוֶנֶת לַמְשִׂימָה: מְכַוְנִים אֶת הָעֲבוֹדָה לַיַעַד, נוֹתְנִים הוֹרָאוֹת, מַדְגִישִׁים תִכְנוּן וְדַדְלִיָנִים, וּמְסַפְּקִים לוּחַ זְמַנִים מְפוֹרָשׁ.

ש: הַמוֹדֵל הַשְׁלִישִׁי: נְתִיב מַטָרָה.

א: אֶוַנְס וְהָאוּס, שְׁנוֹת הַשִׁבְעִים. הַמַנְהִיג מַתְאִים אֶת הַסִגְנוֹן שֶׁלוֹ לַיַעַד, לַתְנָאִים וְלִמְשִׂימָה. וְאַרְבָּעָה סִגְנוֹנוֹת.

ש: הָרִאשׁוֹן.

א: מַנְהִיגוּת תוֹמֶכֶת. דוֹאֵג לִכְפִיפִים, נָגִישׁ, פָּתוּחַ וִידִידוּתִי, יוֹצֵר אַקְלִים שֶׁל צֶוֶת וּמִתְיַחֵס לַכְּפִיפִים כְּשָׁוִים. הַדֻגְמָה: מְפַקֵד גְדוּד אַחֲרֵי פְּעִילוּת קָשָׁה.

ש: הַשֵׁנִי.

א: מַנְהִיגוּת דִירֶקְטִיבִית. אוֹמֵר לַכְּפִיפִים בְּדִיוּק מָה לַעֲשׂוֹת: תִכְנוּן, יַעֲדֵי בִּצוּעַ וּסְטַנְדַרְטִים שֶׁל הִתְנַהֲגוּת.

ש: הַשְׁלִישִׁי.

א: מַנְהִיגוּת מְשַׁתֶפֶת. מִתְיַעֵץ עִם הַכְּפִיפִים עַל הַהַחְלָטוֹת. הַדֻגְמָה: מְנַהֵל תִיכוֹן שֶׁאָמַר לַמוֹרִים שֶׁלוֹ, בַּכִּתָה וּבַמִקְצוֹעַ שֶׁלְךָ אַתָה מַחֲלִיט מָה, אֵיךְ וְכַמָה.

ש: וְהָרְבִיעִי.

א: מַנְהִיגוּת הֶשֵׂגִית. מַצִיב לַכְּפִיפִים יְעָדִים מְאַתְגְרִים, וּמַדְגִישׁ בִּצוּעִים בְּאֵיכוּת גְבוֹהָה. הַדֻגְמָה: חֶבְרוֹת בִּטוּחַ.

ש: וּמָה הַמַסְקָנָה שֶׁל הַמוֹדֵל?

א: מְנַהֵל טוֹב יוֹדֵעַ לִהְיוֹת וֶרְסָטִילִי. הוּא מְשַׁנֶה סִגְנוֹן לְפִי הַמַצָב, הַתְנָאִים, הַסְבִיבָה וְהַכְּפִיפִים.

ש: הַמוֹדֵל הָרְבִיעִי: אֶל אֶם אֶקְס. חִלוּפֵי מַנְהִיג וְחָבֵר.

א: הַמַנְהִיגוּת הִיא תַהֲלִיךְ שֶׁמִתְמַקֵד בָּאִינְטֶרְאַקְצִיָה בֵּין הַמַנְהִיג לַכְּפִיפִים. בְּמֶרְכַּז הַתַהֲלִיךְ עוֹמְדִים הַיְחָסִים הַדִיאָדִיִים, כְּלוֹמַר הַיְחָסִים הַדוּ כִּיווּנִיִים בֵּין הַמַנְהִיג לְכֹל כָּפִיף בְּנִפְרָד.

ש: וּמָה חָדָשׁ בּוֹ?

א: לְדִבְרֵי הַמַרְצָה, זוֹ הַפַּעַם הָרִאשׁוֹנָה שֶׁגִישַׁת מַנְהִיגוּת מִסְתַכֶּלֶת גַם עַל הַמֻנְהָגִים. וְהַמִלָה הַמֶרְכָּזִית הִיא הֲדָדִיוּת: הַמַנְהִיג גַם נוֹתֵן וְגַם מְקַבֵּל מִכֹּל כָּפִיף.

ש: קְבוּצַת הַפָּנִים וּקְבוּצַת הַחוּץ.

א: קְבוּצַת הַפָּנִים מְקַבֶּלֶת מֵהַמַנְהִיג יוֹתֵר מֵידָע, יוֹתֵר הַשְׁפָּעָה, יוֹתֵר אֵמוּן וְיוֹתֵר דְאָגָה. הַחֲבֵרִים בָּה מְעֹרָבִים וְתִקְשָׁרְתִיִים יוֹתֵר, וְעוֹשִׂים מֵעֵבֶר לְמָה שֶׁכָּתוּב בַּחוֹזֶה.

א: קְבוּצַת הַחוּץ פָּחוֹת מַתְאִימָה לְמַנְהִיג. בַּמִלִים שֶׁל הַשֶׁקֶף: בָּאִים לָעֲבוֹדָה, עוֹשִׂים אֶת הָעֲבוֹדָה, וְהוֹלְכִים הַבַּיְתָה. וְהַשְׁאִיפָה שֶׁל כֹּל מְנַהֵל הִיא לְהַרְחִיב אֶת קְבוּצַת הַפָּנִים.

ש: וּמָה הַתוֹצָאוֹת שֶׁל יְחָסִים אֵיכוּתִיִים?

א: פָּחוֹת תַחֲלוּפָה. הַעֲרָכוֹת בִּצוּעַ חִיוּבִיוֹת יוֹתֵר. יוֹתֵר קִדוּמִים. יוֹתֵר מְחֻיָבוּת לָאִרְגוּן. עֲמָדוֹת טוֹבוֹת יוֹתֵר כְּלַפֵּי הָעֲבוֹדָה. יוֹתֵר תְשׂוּמֶת לֵב וּתְמִיכָה מֵהַמַנְהִיג. יוֹתֵר הִשְׁתַתְפוּת. וְקַרְיֵרָה מְהִירָה יוֹתֵר.

ש: וּמִבְּחִינַת הָעוֹבֵד?

א: לְפִי הֶרִיס וַעֲמִיתִיו, בִּיחָסִים טוֹבִים הָעוֹבֵד מְקַבֵּל יַחַס מָעֳדָף, יוֹתֵר תִקְשֹׁרֶת בְּנוֹשְׂאֵי עֲבוֹדָה, יוֹתֵר מָשׁוֹב עַל הַבִּצוּעִים, וְגִישָׁה נוֹחָה לַמְמֻנֶה. וּבִיחָסִים חַלָשִׁים יֵשׁ לוֹ אֵמוּן וּתְמִיכָה מֻגְבָּלִים, וּמְעַט הֲטָבוֹת מֵעֵבֶר לַחוֹזֶה.

ש: וְהַדֻגְמָה מִנֶטְפְלִיקְס?

א: הַחֶבְרָה בִּטְלָה אֶת מִכְסַת יְמֵי הַחֻפְשָׁה, עִם יְעָדִים בְּרוּרִים. הָעוֹבְדִים לֹא לָקְחוּ יוֹתֵר חֹפֶשׁ מִבְּעָבָר, וְהַבִּצוּעִים רַק עָלוּ. זֶה הָרַעֲיוֹן שֶׁל מִמְנַהֲלִים לִמְאַמְנִים.

ש: הַמוֹדֵל הַחֲמִישִׁי: אִינְטֵלִיגֶנְצִיָה רִגְשִׁית.

א: לְפִי הַשֶׁקֶף, זֶה שִׁלוּב שֶׁל כִּשׁוּרִים וִיכוֹלוֹת לֹא קוֹגְנִיטִיבִיִים, שֶׁמַשְׁפִּיעִים עַל הַיְכֹלֶת לְהִתְמוֹדֵד עִם דְרִישׁוֹת וּלְחָצִים.

א: וְדָנִיאֵל גוֹלְמָן מַגְדִיר אוֹתָה כָּךְ: הַיְכֹלֶת לְזַהוֹת אֶת הָרְגָשׁוֹת שֶׁלָנוּ וְשֶׁל אֲחֵרִים, לְהָנִיעַ אֶת עַצְמֵנוּ, וּלְנַהֵל רְגָשׁוֹת הֵיטֵב.

ש: וְלָמָה זֶה חָשׁוּב לַמְנַהֲלִים?

א: כִּי הַמְהַנְדְסִים עִם הָעִפָּרוֹן הֲכִי מְחֻדָד הֵם לֹא בְּהֶכְרֵחַ הַמְנַהֲלִים הֲכִי טוֹבִים. מִי שֶׁיֵשׁ לוֹ אִינְטֵלִיגֶנְצִיָה רִגְשִׁית גְבוֹהָה מַצְלִיחַ יוֹתֵר, גַם אִם הַצִיוּנִים שֶׁלוֹ הָיוּ נְמוּכִים יוֹתֵר.

ש: חֲמֵשֶׁת הַמְמַדִים. הָרִאשׁוֹן.

א: מוּדָעוּת עַצְמִית: לְזַהוֹת אֶת מַצְבֵי הָרוּחַ וְהָרְגָשׁוֹת שֶׁלָךְ, וְאֵיךְ הֵם מַשְׁפִּיעִים עַל אֲחֵרִים. הַדֻגְמָה: אֲחוֹתָה שֶׁל הַמַרְצָה יוֹדַעַת שֶׁכְּשֶׁהִיא רְעֵבָה, אֵין עִם מִי לְדַבֵּר.

ש: הַשֵׁנִי.

א: נִהוּל עַצְמִי: לִשְׁלֹט בַּדְחָפִים וּבְמַצְבֵי הָרוּחַ, וְלַחֲשֹׁב לִפְנֵי שֶׁפּוֹעֲלִים. בַּסְלֵנְג: אֹרֶךְ הַפְּתִיל, לָדַעַת לִסְפֹּר עַד עֶשֶׂר.

ש: הַשְׁלִישִׁי.

א: מוֹטִיבַצְיָה עַצְמִית: לִפְעֹל בִּשְׁבִיל הַיַעַד עַצְמוֹ, וְלֹא בִּשְׁבִיל תַגְמוּל. וְלָכֵן, לְדִבְרֵי הַמַרְצָה, מִתְנַדְבִים הֵם הַרְבֵּה פְּעָמִים יוֹתֵר טוֹבִים.

ש: הָרְבִיעִי וְהַחֲמִישִׁי.

א: אֶמְפַּתְיָה: לְזַהוֹת וּלְהָבִין אֶת הַמְנִיעִים שֶׁל אֲחֵרִים, לִקְרֹא אֲנָשִׁים. וְכִשׁוּרִים חֶבְרָתִיִים: לְנַהֵל מַעַרְכוֹת יְחָסִים וְלִבְנוֹת רִשְׁתוֹת קְשָׁרִים. וְשִׂימוּ לֵב: יְכֹלֶת קוֹגְנִיטִיבִית אֵינָה אֶחָד הַמְמַדִים.

ש: וְאֵיךְ מְשַׁפְּרִים אִינְטֵלִיגֶנְצִיָה רִגְשִׁית?

א: תֹכְנִית הָאִמוּן בַּשֶׁקֶף בְּנוּיָה מִשָׁלוֹשׁ רָמוֹת: מוּדָעוּת עַצְמִית, נִהוּל רְגָשׁוֹת, וּבְנִיַת מַעַרְכוֹת יְחָסִים.

א: וּבַטִיפִּים שֶׁל פוֹרְבֶּס: לַעֲצֹר לִפְנֵי שֶׁמְגִיבִים, לְהַקְשִׁיב, לְשַׁבֵּחַ, לִלְמֹד מִבִּקֹרֶת, לָשִׂים לֵב לִשְׂפַת הַגוּף, לְהִתְנַצֵל, וּלְנַסוֹת לִרְאוֹת מִנְקֻדַת הַמַבָּט שֶׁל הָאַחֵר. הַמַרְצָה הִמְחִישָׁה אֶת זֶה בְּסִפְרָה שֵׁשׁ, שֶׁמֵהַצַד הַשֵׁנִי נִרְאֵית תֵשַׁע.

ש: שְׁאֵלוֹת לַדֶרֶךְ. רִאשׁוֹנָה: מִי עוֹשֶׂה אֶת הַדְבָרִים הַנְכוֹנִים, וּמִי עוֹשֶׂה אֶת הַדְבָרִים נָכוֹן?

~4

א: הַמַנְהִיגִים עוֹשִׂים אֶת הַדְבָרִים הַנְכוֹנִים. הַמְנַהֲלִים עוֹשִׂים אֶת הַדְבָרִים נָכוֹן.

ש: שְׁנִיָה: אֵיזוֹ תְכוּנַת מַנְהִיגוּת הֲכִי קַל לְשַׁפֵּר הַיוֹם?

~4

א: יֶדַע בָּעֵסֶק, כִּי הַיֶדַע זָמִין לְכֻלָם.

ש: שְׁלִישִׁית: עַל הַשָׂרִיג, מִי גָבוֹהַ בִּדְאָגָה לַאֲנָשִׁים וְנָמוּךְ בַּמְשִׂימָה?

~4

א: מְנַהֵל הַקַאנְטְרִי קְלַאבּ.

ש: רְבִיעִית: מְנַהֵל תִיכוֹן אוֹמֵר לַמוֹרִים, בַּכִּתָה שֶׁלְךָ אַתָה מַחֲלִיט מָה, אֵיךְ וְכַמָה. אֵיזֶה סִגְנוֹן זֶה בִּנְתִיב מַטָרָה?

~4

א: מַנְהִיגוּת מְשַׁתֶפֶת.

ש: חֲמִישִׁית: מָה עוֹשִׂים חַבְרֵי קְבוּצַת הַפָּנִים, שֶׁחַבְרֵי קְבוּצַת הַחוּץ לֹא עוֹשִׂים?

~4

א: הֵם עוֹשִׂים מֵעֵבֶר לְמָה שֶׁכָּתוּב בַּחוֹזֶה, וּמְקַבְּלִים יוֹתֵר מֵידָע, הַשְׁפָּעָה, אֵמוּן וּדְאָגָה.

ש: וְאַחֲרוֹנָה: לִסְפֹּר עַד עֶשֶׂר לִפְנֵי שֶׁמְגִיבִים. אֵיזֶה מֵמַד שֶׁל אִינְטֵלִיגֶנְצִיָה רִגְשִׁית זֶה?

~4

א: נִהוּל עַצְמִי.

ש: זֶהוּ לַפֶּרֶק הַזֶה. בַּפֶּרֶק הַבָּא: בַּקָרָה. תַהֲלִיךְ הַבַּקָרָה, שְׁלֹשֶׁת הָעִתוּיִים, וַחֲמֵשֶׁת סוּגֵי הַבַּקָרָה.
